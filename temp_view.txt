    <script>
        document.addEventListener("DOMContentLoaded", function() {
            const startYear = 1993;
            const currentYear = new Date().getFullYear();
            const yearsValue = currentYear - startYear;
            
            const yearsCountElement = document.getElementById('years-count');
            
            if(yearsCountElement) {
                yearsCountElement.setAttribute('data-target', yearsValue);
            }

            const statSection = document.querySelector('.donut-infographic-section');
            let initialAnimated = false;

            const runCounterAnimation = (counter) => {
                if (counter.dataset.animating === "true") return; // Prevent overlapping animations
                counter.dataset.animating = "true";
                
                const target = +counter.getAttribute('data-target');
                const duration = 1000; // 1 second
                const increment = target / (duration / 16);
                let current = 0;

                const animate = () => {
                    current += increment;
                    if(current < target) {
                        counter.textContent = Math.ceil(current);
                        requestAnimationFrame(animate);
                    } else {
                        counter.textContent = target;
                        counter.dataset.animating = "false";
                    }
                };
                animate();
            };

            const observer = new IntersectionObserver((entries) => {
                if(entries[0].isIntersecting && !initialAnimated) {
                    initialAnimated = true;
                    document.querySelectorAll('.count-up').forEach(runCounterAnimation);
                }
            }, { threshold: 0.5 });

            if(statSection) {
                observer.observe(statSection);
            }

            // Interactive Hover & Click re-trigger logic (Desktop)
            const slices = document.querySelectorAll('.stat-slice');
            const textBlocks = document.querySelectorAll('.desktop-svg-wrapper .svg-text-block');
            
            slices.forEach(slice => {
                const index = slice.getAttribute('data-index');
                const triggerInteraction = () => {
                    if (textBlocks[index]) textBlocks[index].classList.add('highlight-text');
                    const counter = slice.querySelector('.count-up');
                    if (counter) runCounterAnimation(counter);
                };
                
                slice.addEventListener('mouseenter', triggerInteraction);
                slice.addEventListener('touchstart', triggerInteraction, {passive: true});
                slice.addEventListener('mouseleave', () => {
                    if (textBlocks[index]) textBlocks[index].classList.remove('highlight-text');
                });
            });


        });
    </script>
